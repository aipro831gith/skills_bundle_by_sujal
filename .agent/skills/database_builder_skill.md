```prisma
// 2_MAIN_CODING_FILES/database/schema.prisma
// Generated by Agent 07 (database_builder) for {product_name}

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ── AUTH ──────────────────────────────────────────────────────────────────

model User {
  id           String    @id @default(uuid())
  email        String    @unique
  passwordHash String    // Argon2id hash — NEVER plain text
  role         Role      @default(USER)
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  clients      Client[]
  invoices     Invoice[]
  sessions     Session[]

  @@index([email])
}

model Session {
  id        String   @id @default(uuid())
  userId    String
  token     String   @unique  // Hashed refresh token
  expiresAt DateTime
  createdAt DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([token])
}

enum Role {
  USER
  ADMIN
  SUPERADMIN
}

// ── CLIENTS ───────────────────────────────────────────────────────────────

model Client {
  id        String    @id @default(uuid())
  userId    String
  name      String
  email     String
  company   String?
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt

  user      User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  invoices  Invoice[]

  @@index([userId])         // ← REQUIRED: FK must have explicit index
  @@unique([userId, email]) // ← Prevent duplicate client email per user
}

// ── INVOICES (FINANCIAL — Int for all money) ──────────────────────────────

model Invoice {
  id             String        @id @default(uuid())
  userId         String
  clientId       String
  totalCents     Int           // ← ALWAYS Int. NEVER Float or Decimal.
  subtotalCents  Int           // ← Sum of line items before GST
  gstCents       Int           // ← GST amount = totalCents - subtotalCents
  gstRatePercent Int           // ← e.g., 3 means 3%
  status         InvoiceStatus @default(DRAFT)
  notes          String?
  dueDate        DateTime?
  paidAt         DateTime?
  createdAt      DateTime      @default(now())
  updatedAt      DateTime      @updatedAt

  user           User          @relation(fields: [userId], references: [id])
  client         Client        @relation(fields: [clientId], references: [id])
  lineItems      LineItem[]
  payments       Payment[]

  @@index([userId])
  @@index([clientId])
  @@index([status])
  @@index([userId, status])   // ← Composite: common dashboard query pattern
}

model LineItem {
  id            String  @id @default(uuid())
  invoiceId     String
  description   String
  quantity      Int     // ← Positive integer only
  unitPriceCents Int    // ← ALWAYS Int cents
  totalCents    Int     // ← = quantity × unitPriceCents (denormalized for query performance)

  invoice       Invoice @relation(fields: [invoiceId], references: [id], onDelete: Cascade)

  @@index([invoiceId])        // ← REQUIRED: FK index
}

model Payment {
  id            String        @id @default(uuid())
  invoiceId     String
  amountCents   Int           // ← Integer cents
  method        PaymentMethod
  reference     String?       // External transaction ID
  paidAt        DateTime      @default(now())

  invoice       Invoice       @relation(fields: [invoiceId], references: [id])

  @@index([invoiceId])        // ← REQUIRED: FK index
}

enum InvoiceStatus {
  DRAFT
  SENT
  PAID
  OVERDUE
  CANCELLED
}

enum PaymentMethod {
  BANK_TRANSFER
  CARD
  CASH
  OTHER
}
```

### Artifact 2: PrismaClient Singleton

```typescript
// 2_MAIN_CODING_FILES/database/prismaClient.ts
// Singleton pattern — prevents connection pool exhaustion
import { PrismaClient } from '@prisma/client';

const globalForPrisma = global as unknown as { prisma?: PrismaClient };

export const prisma =
  globalForPrisma.prisma ??
  new PrismaClient({
    log: process.env.NODE_ENV === 'development' ? ['query', 'warn', 'error'] : ['error'],
    datasources: {
      db: {
        url: process.env.DATABASE_URL,  // connection_limit=10 set in URL param
      },
    },
  });

if (process.env.NODE_ENV !== 'production') {
  globalForPrisma.prisma = prisma;
}

// Graceful shutdown
process.on('beforeExit', async () => {
  await prisma.$disconnect();
});
```

### Artifact 3: Migration Workflow (Forward + Backward)

**Forward migration** (always generated by Prisma):
```bash
npx prisma migrate dev --name "create_invoices_table"
# Creates: prisma/migrations/{timestamp}_create_invoices_table/migration.sql
```

**Backward/rollback migration** (MANUALLY written alongside every forward migration):
```sql
-- prisma/migrations/{timestamp}_create_invoices_table/rollback.sql
-- ROLLBACK: undo the create_invoices_table migration

DROP TABLE IF EXISTS "Payment" CASCADE;
DROP TABLE IF EXISTS "LineItem" CASCADE;
DROP TABLE IF EXISTS "Invoice" CASCADE;
DROP TABLE IF EXISTS "Client" CASCADE;
DROP TYPE IF EXISTS "InvoiceStatus";
DROP TYPE IF EXISTS "PaymentMethod";
```

**Migration verification command:**
```bash
# Verify no pending migrations before deployment
npx prisma migrate status 2>&1 | grep -q "Database schema is up to date" \
  && echo "MIGRATIONS:PASS" \
  || { echo "MIGRATIONS:FAIL — pending migrations exist"; exit 1; }
```

### Artifact 4: Double-Entry Invariant Check

For apps handling payments — verify Invoice totals are consistent:

```typescript
// 2_MAIN_CODING_FILES/database/invariantChecks.ts
// Run as a scheduled job or in dev to verify data integrity

export async function verifyInvoiceTotalsInvariant(): Promise<void> {
  const invoices = await prisma.invoice.findMany({
    include: { lineItems: true, payments: true },
  });

  const violations: string[] = [];

  for (const invoice of invoices) {
    // Invariant 1: totalCents = subtotalCents + gstCents
    if (invoice.totalCents !== invoice.subtotalCents + invoice.gstCents) {
      violations.push(`INVARIANT_FAIL: Invoice ${invoice.id} — total mismatch`);
    }

    // Invariant 2: subtotalCents = SUM(lineItems.totalCents)
    const lineItemSum = invoice.lineItems.reduce((s, li) => s + li.totalCents, 0);
    if (invoice.subtotalCents !== lineItemSum) {
      violations.push(`INVARIANT_FAIL: Invoice ${invoice.id} — subtotal mismatch (stored: ${invoice.subtotalCents}, computed: ${lineItemSum})`);
    }

    // Invariant 3: lineItem.totalCents = quantity × unitPriceCents
    for (const li of invoice.lineItems) {
      if (li.totalCents !== li.quantity * li.unitPriceCents) {
        violations.push(`INVARIANT_FAIL: LineItem ${li.id} — total mismatch`);
      }
    }

    // Invariant 4: PAID invoices must have payment records summing to totalCents
    if (invoice.status === 'PAID') {
      const paidSum = invoice.payments.reduce((s, p) => s + p.amountCents, 0);
      if (paidSum < invoice.totalCents) {
        violations.push(`INVARIANT_FAIL: Invoice ${invoice.id} — marked PAID but payments (${paidSum}¢) < total (${invoice.totalCents}¢)`);
      }
    }
  }

  if (violations.length > 0) {
    throw new Error(`Double-entry invariant violations:\n${violations.join('\n')}`);
  }
}
```

## 5. STRICT CONSTRAINTS

- **NEVER** use `Float` or `Decimal` for monetary fields — `Int` ONLY.
- **NEVER** add a foreign key without a corresponding `@@index`.
- **NEVER** create a migration without also writing the rollback script.
- **NEVER** create a new PrismaClient outside the singleton file.
- **NEVER** store plain-text passwords — `passwordHash` field, Argon2id always.
- **NEVER** store actual refresh token strings — store the hashed token only.

## 7. ESCALATION MATRIX

| Error | Severity | Action |
|-------|----------|--------|
| Float/Decimal money field found | CRITICAL | Change to `Int`. Write migration to `ALTER COLUMN ... TYPE INTEGER`. |
| FK without @@index | HIGH | Add `@@index([fieldName])`. Run `prisma migrate dev`. |
| No rollback.sql for a migration | HIGH | Write rollback before marking task complete. |
| Prisma validate fails | CRITICAL | Fix schema errors. Do not run migrations with invalid schema. |
| Double-entry invariant fails in dev | HIGH | Investigate and fix data calculation logic in mathService. |
| PrismaClient not singleton | HIGH | Refactor to singleton. Connection pool limit must not be per-request. |
